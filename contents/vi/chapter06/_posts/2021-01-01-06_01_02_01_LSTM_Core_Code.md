## 4. Đoạn Code

Hãy triển khai LSTM từ đầu để hiểu mọi thao tác, sau đó so sánh với phiên bản tối ưu của PyTorch:

```python
import numpy as np

class LSTMCell:
    """
    Một LSTM cell triển khai tính toán của một bước thời gian.
    
    Bản triển khai này làm tường minh mọi thao tác. Framework hiện đại
    hợp nhất các thao tác này để hiệu năng, nhưng mục tiêu của ta là hiểu,
    không phải tốc độ. Ta sẽ thấy chính xác cổng điều tiết luồng thông tin ra sao.
    """
    
    def __init__(self, input_size, hidden_size):
        """
        Khởi tạo LSTM cell với khởi tạo trọng số cẩn thận.
        
        Ta dùng khởi tạo Xavier/Glorot scale cho sigmoid và tanh.
        Lược đồ khởi tạo này được thiết kế đặc biệt để duy trì
        scale kích hoạt và gradient hợp lý, ngăn cả biến mất
        lẫn bùng nổ sớm trong huấn luyện.
        """
        self.input_size = input_size
        self.hidden_size = hidden_size
        
        # Chiều đầu vào kết hợp (ẩn + đầu vào)
        combined_size = hidden_size + input_size
        
        # Khởi tạo trọng số cho bốn cổng
        # Dùng khởi tạo Xavier: scale theo 1/√(combined_size)
        scale = 1.0 / np.sqrt(combined_size)
        
        self.Wf = np.random.randn(hidden_size, combined_size) * scale
        self.Wi = np.random.randn(hidden_size, combined_size) * scale
        self.WC = np.random.randn(hidden_size, combined_size) * scale
        self.Wo = np.random.randn(hidden_size, combined_size) * scale
        
        # Khởi tạo bias
        # Bias cổng forget thường khởi tạo = 1 để khuyến khích nhớ ban đầu
        # Đây gọi là "forget bias trick" — bắt đầu bằng cách nhớ mọi thứ
        self.bf = np.ones((hidden_size, 1))  # Bắt đầu với cổng forget cao!
        self.bi = np.zeros((hidden_size, 1))
        self.bC = np.zeros((hidden_size, 1))
        self.bo = np.zeros((hidden_size, 1))
    
    def sigmoid(self, x):
        """Sigmoid ổn định số"""
        return np.where(x >= 0,
                       1 / (1 + np.exp(-x)),
                       np.exp(x) / (1 + np.exp(x)))
    
    def forward(self, x_t, h_prev, c_prev):
        """
        Một bước thời gian LSTM.
        
        x_t: đầu vào hiện tại (input_size, 1)
        h_prev: trạng thái ẩn trước (hidden_size, 1)
        c_prev: trạng thái tế bào trước (hidden_size, 1)
        
        Returns:
            h_t: trạng thái ẩn mới
            c_t: trạng thái tế bào mới
            gates: dictionary các giá trị cổng (để phân tích/debug)
        """
        # Nối trạng thái ẩn trước và đầu vào hiện tại
        # Vectơ kết hợp này ảnh hưởng mọi cổng
        combined = np.vstack([h_prev, x_t])
        
        # Tính tất cả cổng
        # Mỗi cổng là hàm sigmoid học được của đầu vào kết hợp
        f_t = self.sigmoid(self.Wf @ combined + self.bf)  # Cổng forget
        i_t = self.sigmoid(self.Wi @ combined + self.bi)  # Cổng input
        C_tilde = np.tanh(self.WC @ combined + self.bC)  # Giá trị ứng viên
        o_t = self.sigmoid(self.Wo @ combined + self.bo)  # Cổng output
        
        # Cập nhật trạng thái tế bào: quên cũ, thêm mới
        # Đây là phương trình THEN CHỐT của LSTM
        # Lưu ý: phi tuyến tối thiểu, chủ yếu tổ hợp tuyến tính
        c_t = f_t * c_prev + i_t * C_tilde
        
        # Tính trạng thái ẩn mới từ trạng thái tế bào
        # tanh nén trạng thái tế bào không giới hạn về [-1, 1]
        # cổng output điều tiết những gì được phơi bày
        h_t = o_t * np.tanh(c_t)
        
        # Trả về trạng thái và cổng (cổng hữu ích để trực quan hóa/debug)
        gates = {
            'forget': f_t,
            'input': i_t,
            'output': o_t,
            'candidate': C_tilde
        }
        
        return h_t, c_t, gates


class LSTM:
    """
    Mạng LSTM đầy đủ cho xử lý chuỗi.
    
    Bọc LSTMCell để xử lý toàn bộ chuỗi, duy trì trạng thái qua
    các bước thời gian. Đây là LSTM hoàn chỉnh như dùng trong thực tế.
    """
    
    def __init__(self, input_size, hidden_size, output_size):
        self.cell = LSTMCell(input_size, hidden_size)
        self.hidden_size = hidden_size
        
        # Chiếu đầu ra (ánh xạ trạng thái ẩn sang dự đoán)
        self.Why = np.random.randn(output_size, hidden_size) * 0.01
        self.by = np.zeros((output_size, 1))
    
    def forward(self, inputs, return_sequences=False):
        """
        Xử lý toàn bộ chuỗi.
        
        inputs: list các vectơ đầu vào [x_1, ..., x_T]
        return_sequences: nếu True, trả về đầu ra ở mọi bước thời gian
                         nếu False, chỉ trả về đầu ra cuối
        
        Returns:
            outputs: dự đoán ở mỗi bước thời gian (nếu return_sequences=True)
                     hoặc chỉ dự đoán cuối (nếu False)
            hidden_states: tất cả trạng thái ẩn [h_0, ..., h_T]
            cell_states: tất cả trạng thái tế bào [c_0, ..., c_T]
            all_gates: giá trị cổng ở mỗi bước thời gian (để phân tích)
        """
        # Khởi tạo trạng thái về zero
        h = np.zeros((self.hidden_size, 1))
        c = np.zeros((self.hidden_size, 1))
        
        # Theo dõi tiến hóa qua thời gian
        hidden_states = [h]
        cell_states = [c]
        outputs = []
        all_gates = []
        
        # Xử lý chuỗi từng bước
        for x_t in inputs:
            # Cập nhật LSTM cell
            h, c, gates = self.cell.forward(x_t, h, c)
            
            hidden_states.append(h)
            cell_states.append(c)
            all_gates.append(gates)
            
            # Tính đầu ra từ trạng thái ẩn
            y_t = self.Why @ h + self.by
            outputs.append(y_t)
        
        if return_sequences:
            return outputs, hidden_states, cell_states, all_gates
        else:
            # Với phân loại chuỗi, chỉ dùng đầu ra cuối
            return outputs[-1], hidden_states, cell_states, all_gates

# Nhớ số đầu, bỏ qua các số giữa, dự đoán first + last
def create_memory_task(n_samples=100, seq_length=20):
    """
    Tác vụ: Cho chuỗi [a, x, x, x, ..., x, b], dự đoán a + b
    
    Điều này đòi hỏi nhớ 'a' qua seq_length-2 giá trị xen giữa,
    đúng loại phụ thuộc tầm xa mà RNN vanilla gặp khó.
    """
    inputs_list = []
    targets_list = []
    
    for _ in range(n_samples):
        # Số đầu và cuối ngẫu nhiên
        first = np.random.rand() * 10
        last = np.random.rand() * 10
        
        # Điền giữa bằng nhiễu
        sequence = [np.array([[first]])]
        for _ in range(seq_length - 2):
            sequence.append(np.array([[np.random.rand() * 10]]))
        sequence.append(np.array([[last]]))
        
        target = np.array([[first + last]])
        
        inputs_list.append(sequence)
        targets_list.append(target)
    
    return inputs_list, targets_list

# Triển khai LSTM PyTorch
import torch
import torch.nn as nn
import torch.optim as optim

class LSTMNetwork(nn.Module):
    """
    LSTM dùng triển khai tối ưu của PyTorch.
    
    LSTM của PyTorch được tối ưu cao, dùng kernel cuDNN trên GPU
    cho hiệu năng tối đa. Nó xử lý tự động mọi tính toán cổng,
    quản lý trạng thái, và backpropagation.
    """
    
    def __init__(self, input_size, hidden_size, output_size, num_layers=1):
        super(LSTMNetwork, self).__init__()
        
        # Module LSTM của PyTorch
        # num_layers > 1 xếp chồng LSTM (đầu ra lớp này làm đầu vào lớp sau)
        # dropout giữa các lớp giúp regularize LSTM xếp chồng
        self.lstm = nn.LSTM(input_size, hidden_size, num_layers,
                           batch_first=True, dropout=0.0 if num_layers == 1 else 0.2)
        
        # Lớp đầu ra
        self.fc = nn.Linear(hidden_size, output_size)
        
        self.hidden_size = hidden_size
        self.num_layers = num_layers
    
    def forward(self, x, hidden=None):
        """
        x: (batch, seq_len, input_size)
        hidden: tuple tùy chọn (h_0, c_0)
        
        Returns:
            output: (batch, seq_len, output_size) nếu return_sequences
                    hoặc (batch, output_size) nếu không
            (h_n, c_n): trạng thái ẩn và tế bào cuối
        """
        # LSTM trả về:
        # - lstm_out: trạng thái ẩn ở mọi bước thời gian
        # - (h_n, c_n): trạng thái ẩn và tế bào cuối cho mọi lớp
        lstm_out, (h_n, c_n) = self.lstm(x, hidden)
        
        # Dùng bước thời gian cuối cho phân loại chuỗi
        # hoặc mọi bước thời gian cho chuỗi-sang-chuỗi
        final_hidden = lstm_out[:, -1, :]  # Bước thời gian cuối
        
        output = self.fc(final_hidden)
        
        return output, (h_n, c_n)

# Chuyển dữ liệu thành tensor
def prepare_torch_data(inputs_list, targets_list):
    """Chuyển list chuỗi thành tensor theo batch"""
    # Tìm độ dài max (để pad)
    max_len = max(len(seq) for seq in inputs_list)
    
    # Pad chuỗi và stack
    X = []
    y = []
    for inputs, target in zip(inputs_list, targets_list):
        # Chuyển chuỗi thành tensor và pad
        seq_tensor = torch.tensor([inp.flatten() for inp in inputs], 
                                  dtype=torch.float32)
        X.append(seq_tensor)
        y.append(torch.tensor(target.flatten(), dtype=torch.float32))
    
    # Stack thành tensor theo batch
    X = torch.stack([s for s in X])  # (n_samples, seq_len, 1)
    y = torch.stack(y)  # (n_samples, 1)
    
    return X, y

X_train, y_train = prepare_torch_data(train_inputs, train_targets)

model = LSTMNetwork(input_size=1, hidden_size=16, output_size=1, num_layers=2)
criterion = nn.MSELoss()

# Vòng huấn luyện
model.train()
for epoch in range(300):
    # Forward pass
    predictions, _ = model(X_train)
    loss = criterion(predictions, y_train)
    
    # Backward pass
    optimizer.zero_grad()
    loss.backward()  # BPTT qua LSTM diễn ra tự động ở đây
    
    # Gradient clipping (thực hành tốt cho RNN/LSTM)
    torch.nn.utils.clip_grad_norm_(model.parameters(), max_norm=5.0)
    
    optimizer.step()
    
    if epoch % 60 == 0:
        print(f"Epoch {epoch:3d}: Loss = {loss.item():.4f}")

# Kiểm thử trên ví dụ mới

    ]
    
    for seq, true_sum in test_cases:
        seq_tensor = torch.tensor(seq, dtype=torch.float32).unsqueeze(0).unsqueeze(-1)

        print(f"  True sum: {true_sum:.1f}, Predicted: {pred.item():.2f}")
        print(f"  Error: {abs(pred.item() - true_sum):.2f}\n")

print("LSTM đã học thành công nhớ giá trị đầu qua nhiều bước!")
print("Điều này minh họa ưu thế so với RNN vanilla cho phụ thuộc tầm xa.")
```
