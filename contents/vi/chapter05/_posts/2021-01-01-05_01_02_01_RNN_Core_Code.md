## 4. Đoạn Code

Hãy triển khai RNN từ những nguyên lý đầu tiên để hiểu mọi chi tiết, sau đó xem các triển khai PyTorch hiện đại:

```python
import numpy as np
import matplotlib.pyplot as plt

class VanillaRNN:
    """
    Triển khai RNN vanilla từ đầu bằng NumPy.
    
    Bản triển khai này ưu tiên độ rõ ràng hơn hiệu năng. Ta sẽ thấy chính xác
    trạng thái ẩn tiến hóa ra sao theo thời gian, gradient chảy ngược thế nào,
    và gradient biến mất xuất phát từ đâu. Hiểu triển khai thủ công này
    rất quan trọng khi debug huấn luyện RNN và để đánh giá những gì framework
    hiện đại làm tự động.
    """
    
    def __init__(self, input_size, hidden_size, output_size, learning_rate=0.01):
        """
        Khởi tạo RNN với khởi tạo trọng số cẩn thận.
        
        Với RNN, khởi tạo còn quan trọng hơn mạng feedforward.
        Trọng số hồi quy Whh quyết định trị riêng của động lực trạng thái ẩn.
        Nếu trị riêng quá lớn (> 1), trạng thái ẩn bùng nổ. Quá nhỏ (< 1),
        chúng biến mất. Ta khởi tạo giá trị ngẫu nhiên nhỏ với hy vọng rơi vào
        chế độ ổn định, dù điều này thường thất bại với RNN vanilla.
        """
        self.hidden_size = hidden_size
        self.learning_rate = learning_rate
        
        # Trọng số đầu vào sang ẩn: ánh xạ không gian đầu vào sang không gian ẩn
        # Scale theo 1/√hidden_size để duy trì phương sai
        self.Wxh = np.random.randn(hidden_size, input_size) * 0.01
        
        # Trọng số ẩn sang ẩn: ma trận quan trọng nhất cho động lực thời gian
        # Một số người khởi tạo identity + nhiễu để luồng gradient ban đầu tốt hơn
        self.Whh = np.random.randn(hidden_size, hidden_size) * 0.01
        
        # Trọng số ẩn sang đầu ra
        self.Why = np.random.randn(output_size, hidden_size) * 0.01
        
        # Bias
        self.bh = np.zeros((hidden_size, 1))
        self.by = np.zeros((output_size, 1))
    
    def forward(self, inputs, h_prev):
        """
        Forward pass qua toàn bộ chuỗi.
        
        inputs: list các vectơ đầu vào [x_1, x_2, ..., x_T], mỗi x_t là (input_size, 1)
        h_prev: trạng thái ẩn ban đầu (hidden_size, 1)
        
        Returns:
            outputs: list các vectơ đầu ra [y_1, y_2, ..., y_T]
            hidden_states: list các trạng thái ẩn [h_0, h_1, ..., h_T]
            
        Ta phải lưu TẤT CẢ trạng thái ẩn vì backpropagation through time
        cần chúng. Điều này tạo chi phí bộ nhớ tuyến tính theo độ dài chuỗi,
        trở nên quá tải với các chuỗi rất dài.
        """
        hidden_states = [h_prev]
        outputs = []
        h = h_prev
        
        # Xử lý chuỗi từng bước một
        # Vòng lặp tuần tự này là không thể tránh trong RNN — ta không thể
        # song song hóa qua thời gian vì h_t phụ thuộc h_{t-1}
        for x_t in inputs:
            # Cập nhật trạng thái ẩn: kết hợp trạng thái trước và đầu vào hiện tại
            # tanh giới hạn kích hoạt trong [-1, 1], ngăn bùng nổ
            # (dù gây bão hòa và gradient biến mất)
            z = self.Whh @ h + self.Wxh @ x_t + self.bh
            h = np.tanh(z)
            hidden_states.append(h)
            
            # Tính đầu ra từ trạng thái ẩn
            # Với phân loại chuỗi, ta chỉ dùng đầu ra cuối
            # Với gán nhãn chuỗi, ta dùng tất cả đầu ra
            y = self.Why @ h + self.by
            outputs.append(y)
        
        return outputs, hidden_states
    
    def backward(self, inputs, outputs, hidden_states, targets):
        """
        Backpropagation Through Time (BPTT).
        
        Đây là nơi RNN trở nên thách thức. Ta phải backpropagate gradient
        không chỉ qua các lớp (như mạng feedforward) mà còn ngược theo thời gian.
        Gradient của loss theo Whh bao gồm đóng góp từ mọi bước thời gian,
        tạo ra tích dài các Jacobian dẫn đến gradient biến mất/bùng nổ.
        """
        T = len(inputs)
        
        # Khởi tạo bộ tích lũy gradient
        # Chúng sẽ tích lũy gradient từ mọi bước thời gian
        dWxh = np.zeros_like(self.Wxh)
        dWhh = np.zeros_like(self.Whh)
        dWhy = np.zeros_like(self.Why)
        dbh = np.zeros_like(self.bh)
        dby = np.zeros_like(self.by)
        
        # Gradient chảy ngược qua thời gian
        # Ban đầu bằng zero, sẽ tích lũy khi ta backprop qua thời gian
        dh_next = np.zeros((self.hidden_size, 1))
        
        # Backpropagate through time (từ T xuống 1)
        for t in reversed(range(T)):
            # Sai số đầu ra tại bước thời gian này
            # Với MSE loss: dy = y_pred - y_true
            dy = outputs[t] - targets[t]
            
            # Gradient của tham số lớp đầu ra
            dWhy += dy @ hidden_states[t+1].T
            dby += dy
            
            # Backprop vào trạng thái ẩn
            # Hai nguồn gradient: từ đầu ra tại bước này (dy)
            # và từ các bước tương lai (dh_next)
            dh = self.Why.T @ dy + dh_next
            
            # Backprop qua phi tuyến tanh
            # tanh'(z) = 1 - tanh^2(z) = 1 - h^2
            # Đạo hàm này tiến về 0 khi |h| tiến về 1 (bão hòa)
            # gây gradient biến mất
            dz = (1 - hidden_states[t+1]**2) * dh
            
            # Gradient của tham số lớp ẩn
            dbh += dz
            dWxh += dz @ inputs[t].T
            dWhh += dz @ hidden_states[t].T
            
            # Truyền gradient sang bước thời gian trước
            # Đây là nơi tích các Jacobian xảy ra
            # dh_next sẽ được dùng ở bước t-1
            dh_next = self.Whh.T @ dz
        
        # Clip gradient để ngăn bùng nổ
        # Đây là một "hack", nhưng cần thiết cho RNN vanilla
        # Ta đang xử lý triệu chứng (gradient lớn) chứ không phải nguyên nhân
        # (bất ổn định cơ bản của động lực hồi quy)
        for grad in [dWxh, dWhh, dWhy, dbh, dby]:
            np.clip(grad, -5, 5, out=grad)
        
        return dWxh, dWhh, dWhy, dbh, dby
    
    def update_parameters(self, dWxh, dWhh, dWhy, dbh, dby):
        """Cập nhật gradient descent chuẩn"""
        self.Wxh -= self.learning_rate * dWxh
        self.Whh -= self.learning_rate * dWhh
        self.Why -= self.learning_rate * dWhy
        self.bh -= self.learning_rate * dbh
        self.by -= self.learning_rate * dby

# Minh họa trên tác vụ dự đoán chuỗi đơn giản
print("="*70)
print("Vanilla RNN: Dự đoán số tiếp theo trong chuỗi")
print("="*70)

# Tác vụ: Cho [1, 2, 3], dự đoán [2, 3, 4]
# Kiểm tra xem RNN có học được mẫu tuần tự đơn giản không
def generate_sequence_data(n_sequences=100, seq_length=10):
    """
    Sinh dữ liệu huấn luyện: chuỗi các số tăng dần
    Đầu vào: [start, start+1, start+2, ...]
    Mục tiêu: [start+1, start+2, start+3, ...]
    """
    sequences_in = []
    sequences_target = []
    
    for _ in range(n_sequences):
        start = np.random.randint(0, 50)
        seq = np.array([start + i for i in range(seq_length)])
        sequences_in.append([np.array([[x]], dtype=float) for x in seq])
        sequences_target.append([np.array([[x]], dtype=float) for x in seq[1:] + [seq[-1]+1]])
    
    return sequences_in, sequences_target

# Tạo RNN
rnn = VanillaRNN(input_size=1, hidden_size=10, output_size=1, learning_rate=0.001)

# Sinh dữ liệu
train_inputs, train_targets = generate_sequence_data(n_sequences=50, seq_length=8)

# Vòng huấn luyện
print("Huấn luyện RNN để dự đoán số tiếp theo...")
losses = []

for epoch in range(200):
    epoch_loss = 0
    
    for inputs, targets in zip(train_inputs, train_targets):
        # Forward pass
        h_prev = np.zeros((rnn.hidden_size, 1))
        outputs, hidden_states = rnn.forward(inputs, h_prev)
        
        # Tính loss (MSE)
        loss = sum((out - tgt)**2 for out, tgt in zip(outputs, targets))
        epoch_loss += np.sum(loss)
        
        # Backward pass
        grads = rnn.backward(inputs, outputs, hidden_states, targets)
        
        # Cập nhật
        rnn.update_parameters(*grads)
    
    losses.append(epoch_loss / len(train_inputs))
    
    if epoch % 40 == 0:
        print(f"Epoch {epoch:3d}: Average Loss = {losses[-1]:.4f}")

# Kiểm thử trên chuỗi mới
print("\n" + "="*70)
print("Kiểm thử trên chuỗi mới: [10, 11, 12, 13, 14, 15, 16, 17]")
print("="*70)

test_input = [np.array([[x]], dtype=float) for x in [10, 11, 12, 13, 14, 15, 16, 17]]
h_test = np.zeros((rnn.hidden_size, 1))
test_outputs, test_hidden = rnn.forward(test_input, h_test)

print("Dự đoán:")
for t, (inp, out) in enumerate(zip(test_input, test_outputs)):
    print(f"  Sau khi thấy {int(inp[0,0])}, dự đoán tiếp: {out[0,0]:.2f} (đúng: {int(inp[0,0])+1})")

print("\nLưu ý: RNN vanilla gặp khó với tác vụ đơn giản này do gradient biến mất!")
print("Điều này thúc đẩy kiến trúc LSTM/GRU sẽ được đề cập ở chương tiếp theo.")
```

Bây giờ hãy xem triển khai PyTorch hiện đại xử lý các chi tiết này tự động:

```python
import torch
import torch.nn as nn
import torch.optim as optim

class RNNSequenceModel(nn.Module):
    """
    RNN hiện đại dùng layer RNN tích hợp của PyTorch.
    
    PyTorch xử lý tính toán hồi quy hiệu quả, unrolling vòng lặp
    trong C++ tối ưu và quản lý gradient tự động. Nhanh hơn nhiều so với
    triển khai NumPy của ta và xử lý batching một cách thanh lịch.
    """
    
    def __init__(self, input_size, hidden_size, output_size, num_layers=1):
        super(RNNSequenceModel, self).__init__()
        
        # Module RNN của PyTorch xử lý tính toán hồi quy
        # num_layers > 1 tạo RNN xếp chồng (đầu ra của lớp này làm đầu vào lớp sau)
        # batch_first=True nghĩa là shape đầu vào là (batch, seq_len, input_size)
        self.rnn = nn.RNN(input_size, hidden_size, num_layers, 
                         batch_first=True, nonlinearity='tanh')
        
        # Lớp đầu ra ánh xạ trạng thái ẩn sang dự đoán
        self.fc = nn.Linear(hidden_size, output_size)
        
        self.hidden_size = hidden_size
        self.num_layers = num_layers
    
    def forward(self, x, hidden=None):
        """
        x: (batch, seq_len, input_size)
        hidden: trạng thái ẩn ban đầu tùy chọn (num_layers, batch, hidden_size)
        
        Returns:
            output: (batch, seq_len, output_size)
            hidden: trạng thái ẩn cuối
        """
        # Nếu không cung cấp trạng thái ẩn ban đầu, RNN khởi tạo về zero
        if hidden is None:
            hidden = torch.zeros(self.num_layers, x.size(0), self.hidden_size)
            if x.is_cuda:
                hidden = hidden.cuda()
        
        # RNN trả về:
        # - rnn_out: trạng thái ẩn ở mọi bước thời gian (batch, seq_len, hidden_size)
        # - hidden: trạng thái ẩn cuối (num_layers, batch, hidden_size)
        rnn_out, hidden = self.rnn(x, hidden)
        
        # Áp dụng lớp đầu ra cho mỗi bước thời gian
        # Reshape thành (batch * seq_len, hidden_size) để tính toán hiệu quả
        batch_size, seq_len, _ = rnn_out.shape
        rnn_out = rnn_out.reshape(-1, self.hidden_size)
        output = self.fc(rnn_out)
        output = output.reshape(batch_size, seq_len, -1)
        
        return output, hidden
```
